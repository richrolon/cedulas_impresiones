        <input type="file" id="frontInput" accept="image/*" capture="camera">
        <canvas id="frontCanvas" class="preview-canvas"></canvas>
        <img id="frontPreview" alt="Frontal">

<script>
    const scanner = new jscanify();
    const { jsPDF } = window.jspdf;
    
    let frontImgData = null;
    let backImgData = null;

    // Función para procesar y recortar la imagen
    async function processImage(input, previewId, canvasId) {
        const file = input.files[0];
        if (!file) return;

        const img = new Image();
        img.src = URL.createObjectURL(file);
        
        img.onload = function() {
            const canvas = document.getElementById(canvasId);
            const ctx = canvas.getContext('2d');
            
            // Detectar y recortar automáticamente la cédula
            const resultCanvas = scanner.extractPaper(img, 850, 540); // Proporción estándar ID-1
            
            const preview = document.getElementById(previewId);
            preview.src = resultCanvas.toDataURL('image/jpeg');
            preview.style.display = 'block';

            if (previewId === 'frontPreview') frontImgData = preview.src;
            else backImgData = preview.src;

            checkReady();
        };
    }

    document.getElementById('frontInput').onchange = (e) => processImage(e.target, 'frontPreview', 'frontCanvas');
    document.getElementById('backInput').onchange = (e) => processImage(e.target, 'backPreview', 'backCanvas');

    function checkReady() {
        if (frontImgData && backImgData) {
            document.getElementById('generateBtn').disabled = false;
        }
    }

    document.getElementById('generateBtn').onclick = function() {
        const doc = new jsPDF({
            orientation: 'portrait',
            unit: 'mm',
            format: 'a4'
        });

        // Medidas estándar de cédula en mm
        const w = 85; 
        const h = 54;
        const margin = 10;
        const yCoord = 20; // Posición vertical superior

        // Colocar imágenes una al lado de la otra
        doc.addImage(frontImgData, 'JPEG', margin, yCoord, w, h);
        doc.addImage(backImgData, 'JPEG', margin + w + 5, yCoord, w, h);

        // Abrir diálogo de impresión
        doc.autoPrint();
        window.open(doc.output('bloburl'), '_blank');
    };
</script>
